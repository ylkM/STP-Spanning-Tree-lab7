! =====================================================
! SW3 Configuration - STP Lab
! =====================================================

enable
configure terminal

hostname SW3

! --- VLANs ---
vlan 2
 name VLAN2
 exit

! --- Trunk to SW1 (Fa0/1) ---
interface FastEthernet0/1
 switchport mode trunk
 switchport trunk allowed vlan 1,2
 exit

! --- Trunk to SW2 (Fa0/2) ---
interface FastEthernet0/2
 switchport mode trunk
 switchport trunk allowed vlan 1,2
 exit

! --- Access port to end device (VLAN1, 172.16.0.0/25) ---
interface FastEthernet0/3
 switchport mode access
 switchport access vlan 1
 exit

! =====================================================
! STEP 5: PortFast + BPDU Guard on the access (end-device) port
!         Never enable these on trunk/inter-switch ports.
! =====================================================
interface FastEthernet0/3
 spanning-tree portfast
 spanning-tree bpduguard enable
 exit

end
write memory
